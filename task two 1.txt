TITLE Student Performance Analytics System (Refactored with STRUCT)

INCLUDE Irvine32.inc

MAX_SIZE equ 100
EXC_VAL  equ 1
AVG_VAL  equ 2
RISK_VAL equ 3

Student STRUCT
    StudentID   DWORD ?
    Attendance  DWORD ?
    QuizMarks   DWORD ?
    AssignMarks DWORD ?
    LabScore    DWORD ?
    Category    DWORD ?
Student ENDS

.data
    studentArr      Student MAX_SIZE DUP(<>)
    totalRecords    DWORD 0

    msgMenu         BYTE "        STUDENT PERFORMANCE ANALYTICS SYSTEM        ", 0dh, 0ah
                    BYTE "1. Add New Student Record", 0dh, 0ah
                    BYTE "2. Display Complete Analytics Report", 0dh, 0ah
                    BYTE "3. Search Student by ID", 0dh, 0ah
                    BYTE "4. Filter Students by Category", 0dh, 0ah
                    BYTE "5. Update/Fix Invalid Record", 0dh, 0ah
                    BYTE "6. Exit System", 0dh, 0ah
                    BYTE "Enter your choice (1-6): ", 0

    askID           BYTE "Enter Student ID (Positive Integer): ", 0
    askAtt          BYTE "Enter Attendance Percentage (0-100): ", 0
    askQuiz         BYTE "Enter Quiz Marks (0-100): ", 0
    askAssign       BYTE "Enter Assignment Marks (0-100): ", 0
    askLab          BYTE "Enter Lab Score (0-100): ", 0
    askSearch       BYTE "Enter Student ID to Search: ", 0
    askFilter       BYTE "Enter Category to Filter (1=Excellent, 2=Average, 3=At Risk): ", 0

    errFull         BYTE "Error: Database is full! Cannot add more records.", 0dh, 0ah, 0
    errDup          BYTE "Error: Student ID already exists in the system.", 0dh, 0ah, 0
    errRange        BYTE "Error: Invalid input range. Must be between 0 and 100.", 0dh, 0ah, 0
    errEmpty        BYTE "Error: Database is empty. No records to process.", 0dh, 0ah, 0
    errNoMatch      BYTE "Error: Student record not found.", 0dh, 0ah, 0
    errOverflow     BYTE "Warning: Arithmetic overflow detected during calculations!", 0dh, 0ah, 0
    errBadCat       BYTE "Error: Invalid category chosen.", 0dh, 0ah, 0

    txtHeader       BYTE "ID         | Att% | Quiz | Asgmt | Lab  | Calculated Avg | Category", 0dh, 0ah, 0
    txtExc          BYTE "Excellent", 0
    txtAvg          BYTE "Average", 0
    txtRisk         BYTE "At Risk", 0
    txtSpacer       BYTE "        | ", 0
    txtDivider      BYTE " | ", 0
    msgSuccess      BYTE "Record updated successfully!", 0dh, 0ah, 0
    msgWait         BYTE 0dh, 0ah, "Press any key to return to menu...", 0dh, 0ah, 0

.code
main PROC
    windowLoop:
        call Clrscr
        mov edx, OFFSET msgMenu
        call WriteString
        
        call ReadInt
        call Crlf

        cmp eax, 1
        je  goAdd
        cmp eax, 2
        je  goDisplay
        cmp eax, 3
        je  goSearch
        cmp eax, 4
        je  goFilter
        cmp eax, 5
        je  goUpdate
        cmp eax, 6
        je  goExit
        jmp windowLoop

    goAdd:
        call AddRecord
        jmp menuPause
    goDisplay:
        call DisplayReport
        jmp menuPause
    goSearch:
        call SearchID
        jmp menuPause
    goFilter:
        call FilterCategory
        jmp menuPause
    goUpdate:
        call UpdateRecord
        jmp menuPause
    goExit:
        exit

    menuPause:
        mov edx, OFFSET msgWait
        call WriteString
        call ReadChar
        jmp windowLoop
main ENDP

AddRecord PROC
    pushad

    mov eax, totalRecords
    cmp eax, MAX_SIZE
    jb  spaceOk
    mov edx, OFFSET errFull
    call WriteString
    jmp exitAdd

spaceOk:
    mov edx, OFFSET askID
    call WriteString
    call ReadInt
    mov ebx, eax

    ; --- Check for Duplicates ---
    mov ecx, totalRecords
    lea edi, studentArr
    cmp ecx, 0
    je  id_is_valid
    
checkDup:
    cmp (Student PTR [edi]).StudentID, ebx
    je  dupErr
    add edi, TYPE Student
    dec ecx
    je  id_is_valid    ; Replaces loop constraint
    jmp checkDup       ; Safe far jump

dupErr:
    mov edx, OFFSET errDup
    call WriteString
    jmp exitAdd

id_is_valid:
    ; --- Calculate offset for new record ---
    mov eax, totalRecords
    mov edi, TYPE Student
    mul edi
    lea esi, [studentArr + eax]
    
    mov (Student PTR [esi]).StudentID, ebx

getAtt:
    mov edx, OFFSET askAtt
    call WriteString
    call ReadInt
    cmp eax, 0
    jl  badAtt
    cmp eax, 100
    jg  badAtt
    mov (Student PTR [esi]).Attendance, eax
    jmp getQuiz

badAtt:
    mov edx, OFFSET errRange
    call WriteString
    jmp getAtt

getQuiz:
    mov edx, OFFSET askQuiz
    call WriteString
    call ReadInt
    cmp eax, 0
    jl  badQz
    cmp eax, 100
    jg  badQz
    mov (Student PTR [esi]).QuizMarks, eax
    jmp getAssign

badQz:
    mov edx, OFFSET errRange
    call WriteString
    jmp getQuiz

getAssign:
    mov edx, OFFSET askAssign
    call WriteString
    call ReadInt
    cmp eax, 0
    jl  badAsg
    cmp eax, 100
    jg  badAsg
    mov (Student PTR [esi]).AssignMarks, eax
    jmp getLab

badAsg:
    mov edx, OFFSET errRange
    call WriteString
    jmp getAssign

getLab:
    mov edx, OFFSET askLab
    call WriteString
    call ReadInt
    cmp eax, 0
    jl  badLab
    cmp eax, 100
    jg  badLab
    mov (Student PTR [esi]).LabScore, eax
    jmp saveData

badLab:
    mov edx, OFFSET errRange
    call WriteString
    jmp getLab

saveData:
    mov eax, (Student PTR [esi]).Attendance
    mov ebx, (Student PTR [esi]).QuizMarks
    mov ecx, (Student PTR [esi]).AssignMarks
    mov edx, (Student PTR [esi]).LabScore
    call CalculateCategory
    
    mov (Student PTR [esi]).Category, eax
    inc totalRecords

exitAdd:
    popad
    ret
AddRecord ENDP

CalculateCategory PROC
    push ebx
    push ecx
    push edx
    push esi

    mov esi, eax

    cmp esi, 75
    jl  setRisk

    xor eax, eax
    add eax, ebx
    jo  overflowFlag
    add eax, ecx
    jo  overflowFlag
    add eax, edx
    jo  overflowFlag
    jmp dynamicCalc

overflowFlag:
    mov edx, OFFSET errOverflow
    call WriteString

dynamicCalc:
    mov edx, 0
    mov ebx, 3
    div ebx

    cmp eax, 85
    jae setExc
    cmp eax, 60
    jae setAvg
    jmp setRisk

setExc:
    mov eax, EXC_VAL
    jmp doneEval
setAvg:
    mov eax, AVG_VAL
    jmp doneEval
setRisk:
    mov eax, RISK_VAL

doneEval:
    pop esi
    pop edx
    pop ecx
    pop ebx
    ret
CalculateCategory ENDP

DisplayReport PROC
    pushad
    
    cmp totalRecords, 0
    jne runPrint
    mov edx, OFFSET errEmpty
    call WriteString
    jmp exitRep

runPrint:
    mov edx, OFFSET txtHeader
    call WriteString

    mov ecx, totalRecords
    lea esi, studentArr

printLoop:
    mov eax, (Student PTR [esi]).StudentID
    call WriteDec
    mov edx, OFFSET txtSpacer
    call WriteString

    mov eax, (Student PTR [esi]).Attendance
    call WriteDec
    mov edx, OFFSET txtDivider
    call WriteString

    mov eax, (Student PTR [esi]).QuizMarks
    call WriteDec
    mov edx, OFFSET txtDivider
    call WriteString

    mov eax, (Student PTR [esi]).AssignMarks
    call WriteDec
    mov edx, OFFSET txtDivider
    call WriteString

    mov eax, (Student PTR [esi]).LabScore
    call WriteDec
    mov edx, OFFSET txtDivider
    call WriteString

    push ecx
    mov eax, (Student PTR [esi]).QuizMarks
    add eax, (Student PTR [esi]).AssignMarks
    add eax, (Student PTR [esi]).LabScore
    mov edx, 0
    mov ecx, 3
    div ecx
    call WriteDec
    pop ecx
    
    mov edx, OFFSET txtDivider
    call WriteString

    mov eax, (Student PTR [esi]).Category
    cmp eax, EXC_VAL
    je  showExc
    cmp eax, AVG_VAL
    je  showAvg
    
    mov edx, OFFSET txtRisk
    jmp printLabel

showExc:
    mov edx, OFFSET txtExc
    jmp printLabel
showAvg:
    mov edx, OFFSET txtAvg

printLabel:
    call WriteString
    call Crlf
    
    add esi, TYPE Student
    dec ecx
    je  exitRep        ; Replaces loop constraint
    jmp printLoop      ; Safe far jump

exitRep:
    popad
    ret
DisplayReport ENDP

SearchID PROC
    pushad

    cmp totalRecords, 0
    jne runSearch
    mov edx, OFFSET errEmpty
    call WriteString
    jmp exitFind

runSearch:
    mov edx, OFFSET askSearch
    call WriteString
    call ReadInt
    mov ebx, eax

    mov ecx, totalRecords
    lea esi, studentArr

searchLoop:
    cmp (Student PTR [esi]).StudentID, ebx
    je  foundRow
    add esi, TYPE Student
    dec ecx
    je  searchFailed   ; Replaces loop constraint
    jmp searchLoop     ; Safe far jump

searchFailed:
    mov edx, OFFSET errNoMatch
    call WriteString
    jmp exitFind

foundRow:
    call Crlf
    mov edx, OFFSET txtHeader
    call WriteString
    
    mov eax, (Student PTR [esi]).StudentID
    call WriteDec
    mov edx, OFFSET txtSpacer
    call WriteString
    
    mov eax, (Student PTR [esi]).Attendance
    call WriteDec
    mov edx, OFFSET txtDivider
    call WriteString
    
    mov eax, (Student PTR [esi]).QuizMarks
    call WriteDec
    mov edx, OFFSET txtDivider
    call WriteString
    
    mov eax, (Student PTR [esi]).AssignMarks
    call WriteDec
    mov edx, OFFSET txtDivider
    call WriteString
    
    mov eax, (Student PTR [esi]).LabScore
    call WriteDec
    mov edx, OFFSET txtDivider
    call WriteString
    
    mov eax, (Student PTR [esi]).QuizMarks
    add eax, (Student PTR [esi]).AssignMarks
    add eax, (Student PTR [esi]).LabScore
    mov edx, 0
    mov edi, 3
    div edi
    call WriteDec
    mov edx, OFFSET txtDivider
    call WriteString

    mov eax, (Student PTR [esi]).Category
    cmp eax, EXC_VAL
    je  findExc
    cmp eax, AVG_VAL
    je  findAvg
    mov edx, OFFSET txtRisk
    jmp findDisp
findExc:
    mov edx, OFFSET txtExc
    jmp findDisp
findAvg:
    mov edx, OFFSET txtAvg
findDisp:
    call WriteString
    call Crlf

exitFind:
    popad
    ret
SearchID ENDP

FilterCategory PROC
    pushad

    cmp totalRecords, 0
    jne runFilter
    mov edx, OFFSET errEmpty
    call WriteString
    jmp exitFilt

runFilter:
    mov edx, OFFSET askFilter
    call WriteString
    call ReadInt
    
    cmp eax, 1
    jl  invalidCat
    cmp eax, 3
    jg  invalidCat
    mov ebx, eax
    
    call Crlf
    mov edx, OFFSET txtHeader
    call WriteString

    mov ecx, totalRecords
    lea esi, studentArr
    mov edi, 0

filterLoop:
    cmp (Student PTR [esi]).Category, ebx
    jne nextFilter
    
    inc edi
    mov eax, (Student PTR [esi]).StudentID
    call WriteDec
    mov edx, OFFSET txtSpacer
    call WriteString
    
    mov eax, (Student PTR [esi]).Attendance
    call WriteDec
    mov edx, OFFSET txtDivider
    call WriteString
    
    mov eax, (Student PTR [esi]).QuizMarks
    call WriteDec
    mov edx, OFFSET txtDivider
    call WriteString
    
    mov eax, (Student PTR [esi]).AssignMarks
    call WriteDec
    mov edx, OFFSET txtDivider
    call WriteString
    
    mov eax, (Student PTR [esi]).LabScore
    call WriteDec
    mov edx, OFFSET txtDivider
    call WriteString

    push ecx
    mov eax, (Student PTR [esi]).QuizMarks
    add eax, (Student PTR [esi]).AssignMarks
    add eax, (Student PTR [esi]).LabScore
    mov edx, 0
    mov ecx, 3
    div ecx
    call WriteDec
    pop ecx
    mov edx, OFFSET txtDivider
    call WriteString

    cmp ebx, EXC_VAL
    je  filtExc
    cmp ebx, AVG_VAL
    je  filtAvg
    mov edx, OFFSET txtRisk
    jmp filtDisp
filtExc:
    mov edx, OFFSET txtExc
    jmp filtDisp
filtAvg:
    mov edx, OFFSET txtAvg
filtDisp:
    call WriteString
    call Crlf

nextFilter:
    add esi, TYPE Student
    dec ecx
    je  evalFilterMatch  ; Replaces loop constraint
    jmp filterLoop       ; Safe far jump
    
evalFilterMatch:    
    cmp edi, 0
    jne exitFilt
    mov edx, OFFSET errNoMatch
    call WriteString
    jmp exitFilt

invalidCat:
    mov edx, OFFSET errBadCat
    call WriteString

exitFilt:
    popad
    ret
FilterCategory ENDP

UpdateRecord PROC
    pushad

    cmp totalRecords, 0
    jne runUpdate
    mov edx, OFFSET errEmpty
    call WriteString
    jmp exitUpd

runUpdate:
    mov edx, OFFSET askSearch
    call WriteString
    call ReadInt
    mov ebx, eax 

    mov ecx, totalRecords
    lea esi, studentArr

updateSearch:
    cmp (Student PTR [esi]).StudentID, ebx
    je  foundUpd
    add esi, TYPE Student
    dec ecx
    je  updateFailed   ; Replaces loop constraint
    jmp updateSearch   ; Safe far jump

updateFailed:
    mov edx, OFFSET errNoMatch
    call WriteString
    jmp exitUpd

foundUpd:
modAtt:
    mov edx, OFFSET askAtt
    call WriteString
    call ReadInt
    cmp eax, 0
    jl  badModAtt
    cmp eax, 100
    jg  badModAtt
    mov (Student PTR [esi]).Attendance, eax
    jmp modQuiz
badModAtt:
    mov edx, OFFSET errRange
    call WriteString
    jmp modAtt

modQuiz:
    mov edx, OFFSET askQuiz
    call WriteString
    call ReadInt
    cmp eax, 0
    jl  badModQz
